cat <<EOF | k apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: taint-deployment
spec:
  replicas: 8
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: nginx:1.20.1
        ports:
        - containerPort: 80
EOF

# Control
sudo crictl ps |wc -l
# Worker
sudo crictl ps |wc -l

kubectl delete deployment taint-deployment

kubectl taint nodes worker \
    bubba=value:PreferNoSchedule

kubectl describe node |grep Taint
